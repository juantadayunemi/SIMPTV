# =================================================================
# COMANDOS PARA INICIAR CADA SERVICIO DE TRAFISMART
# =================================================================
# Copia y pega estos comandos en terminales separadas

# ---------------------------------------------------------------
# TERMINAL 1: REDIS (debe estar corriendo primero)
# ---------------------------------------------------------------
# Si ya está corriendo en otra terminal, ignora esto
cd S:\Construccion\SIMPTV\backend\redis
.\redis-server.exe redis.windows.conf


# ---------------------------------------------------------------
# TERMINAL 2: CELERY WORKER
# ---------------------------------------------------------------
cd S:\Construccion\SIMPTV\backend
.\venv\Scripts\activate
celery -A config worker --loglevel=info --pool=solo


# ---------------------------------------------------------------
# TERMINAL 3: DAPHNE (WebSocket para video en tiempo real)
# ---------------------------------------------------------------
cd S:\Construccion\SIMPTV\backend
.\venv\Scripts\activate
daphne -b 0.0.0.0 -p 8001 config.asgi:application


# ---------------------------------------------------------------
# TERMINAL 4: DJANGO RUNSERVER (API REST)
# ---------------------------------------------------------------
cd S:\Construccion\SIMPTV\backend
.\venv\Scripts\activate
python manage.py runserver


# ---------------------------------------------------------------
# TERMINAL 5: FRONTEND (Vite)
# ---------------------------------------------------------------
cd S:\Construccion\SIMPTV\frontend
npm run dev
